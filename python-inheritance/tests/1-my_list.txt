# Tests pour MyList
# Première ligne: importe la classe MyList
>>> MyList = __import__('1-my_list').MyList

# Test 1: Créer une liste vide et afficher
>>> my_list = MyList()
>>> my_list.print_sorted()
[]

# Test 2: Ajouter des éléments et afficher trié
>>> my_list = MyList([1, 3, 2, 5, 4])
>>> my_list.print_sorted()
[1, 2, 3, 4, 5]

# Test 3: Les éléments négatifs
>>> my_list = MyList([1, -5, 3, -2, 0])
>>> my_list.print_sorted()
[-5, -2, 0, 1, 3]

# Test 4: Vérifier que la liste originale n'est pas modifiée
>>> my_list = MyList([3, 1, 2])
>>> my_list.print_sorted()
[1, 2, 3]
>>> print(my_list)
[3, 1, 2]

# Test 5: Une seule valeur
>>> my_list = MyList([42])
>>> my_list.print_sorted()
[42]

# Test 6: Doublets
>>> my_list = MyList([2, 2, 1, 1, 3])
>>> my_list.print_sorted()
[1, 1, 2, 2, 3]

# Test 7: Ajouter un élément après création
>>> my_list = MyList([5, 1, 3])
>>> my_list.append(2)
>>> my_list.print_sorted()
[1, 2, 3, 5]

# Test 8: MyList est une liste
>>> my_list = MyList([1, 2, 3])
>>> isinstance(my_list, list)
True

# Test 9: Nombres grands
>>> my_list = MyList([100, 1, 50, 2])
>>> my_list.print_sorted()
[1, 2, 50, 100]
